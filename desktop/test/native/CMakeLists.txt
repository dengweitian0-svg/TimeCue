cmake_minimum_required(VERSION 3.20)
project(timecue_native_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(d7_native_tests single_instance_test.cpp ../../windows/runner/single_instance.cpp ../../windows/runner/launch_configuration.cpp)
target_include_directories(d7_native_tests PRIVATE ../../windows/runner)
target_compile_definitions(d7_native_tests PRIVATE UNICODE _UNICODE NOMINMAX)
target_compile_options(d7_native_tests PRIVATE /W4 /WX /utf-8 /EHsc)
target_link_libraries(d7_native_tests PRIVATE user32 advapi32 shell32 ole32)
enable_testing()
add_test(NAME d7_single_instance COMMAND d7_native_tests)
set_tests_properties(d7_single_instance PROPERTIES TIMEOUT 20)

add_executable(d8_native_tests managed_process_test.cpp ../../windows/runner/managed_process.cpp ../../windows/runner/launch_configuration.cpp)
target_include_directories(d8_native_tests PRIVATE ../../windows/runner)
target_compile_definitions(d8_native_tests PRIVATE UNICODE _UNICODE NOMINMAX _WIN32_WINNT=0x0A00)
target_compile_options(d8_native_tests PRIVATE /W4 /WX /utf-8 /EHsc)
target_link_libraries(d8_native_tests PRIVATE shell32 ole32)
enable_testing()
add_test(NAME d8_managed_process COMMAND d8_native_tests)
set_tests_properties(d8_managed_process PROPERTIES TIMEOUT 45)
